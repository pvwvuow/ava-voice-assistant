   338→  const N = 52;
   339→  const levels = new Array(N).fill(0.1);
   340→  let t0 = 0, energy = 0.1;
   341→  function rr(c, x, y, w, hgt, r) {
   342→    if (c.roundRect) { c.beginPath(); c.roundRect(x, y, w, hgt, r); return; }
   343→    c.beginPath();
   344→    c.moveTo(x + r, y);
   345→    c.arcTo(x + w, y, x + w, y + hgt, r);
   346→    c.arcTo(x + w, y + hgt, x, y + hgt, r);
   347→    c.arcTo(x, y + hgt, x, y, r);
   348→    c.arcTo(x, y, x + w, y, r);
   349→    c.closePath();
   350→  }
   351→  function frame() {
   352→    t0 += 0.016;
   353→    const target = state === 'listening' ? 0.88 : state === 'processing' ? 0.42 : state === 'success' ? 0.55 : 0.15;
   354→    energy += (target - energy) * 0.05;
   355→    ctx.clearRect(0, 0, W, H);
   356→    const mid = H / 2;
   357→    const gap = 4;
   358→    const bw = Math.max(2, Math.min(4.5, (W - (N - 1) * gap) / N));
   359→    const startX = (W - (N * bw + (N - 1) * gap)) / 2;
   360→    for (let i = 0; i < N; i++) {
   361→      const env = Math.sin((Math.PI * i) / (N - 1));
   362→      let lvl;
   363→      if (micData) {
   364→        /* صدای واقعی میکروفون: تبدیل طیف فرکانسی به ۵۲ میله — همیشه واقعی */
   365→        const bins = Math.floor(micData.length * 0.72);
   366→        const bi = Math.min(micData.length - 1, Math.floor(Math.pow(i / N, 1.55) * bins));
   367→        const raw = micData[bi] / 255;
   368→        lvl = Math.max(0.05, Math.min(1, raw * 1.6 * (0.35 + 0.65 * env)));
   369→      } else {
   370→        const n =
   371→          Math.sin(t0 * 2.1 + i * 0.55) * 0.5 +
   372→          Math.sin(t0 * 3.7 + i * 1.3) * 0.3 +
   373→          Math.sin(t0 * 0.7 + i * 0.21) * 0.2;
   374→        const amp = energy * env * (0.32 + 0.68 * Math.abs(n));
   375→        const jitter = energy > 0.2 ? Math.random() * 0.13 * energy : 0;
   376→        lvl = Math.max(0.04, Math.min(1, amp + jitter));
   377→      }
   378→      levels[i] += (lvl - levels[i]) * 0.25;
   379→      const bh = Math.max(3, levels[i] * (H - 8));
   380→      const g = ctx.createLinearGradient(0, mid - bh / 2, 0, mid + bh / 2);
   381→      g.addColorStop(0, 'rgba(52, 211, 153, 0.95)');
   382→      g.addColorStop(0.5, 'rgba(16, 185, 129, 0.85)');
   383→      g.addColorStop(1, 'rgba(13, 148, 136, 0.9)');
   384→      ctx.fillStyle = g;
   385→      rr(ctx, startX + i * (bw + gap), mid - bh / 2, bw, bh, bw / 2);
   386→      ctx.fill();
   387→    }
   388→    requestAnimationFrame(frame);
   389→  }

   513→      g.gain.setValueAtTime(0.001, ac.currentTime);
   514→      g.gain.exponentialRampToValueAtTime(0.2, ac.currentTime + 0.05);
   515→      g.gain.exponentialRampToValueAtTime(0.001, ac.currentTime + 0.9);

   649→  function makeRec() {
   650→    const r = new SRC();
   651→    r.lang = 'fa-IR';
   652→    r.interimResults = true;
   653→    r.continuous = false;
   654→    r.onresult = (e) => {
   655→      let interim = '', final = '';
   656→      for (let i = e.resultIndex; i < e.results.length; i++) {
   657→        const t = e.results[i][0].transcript;
   658→        if (e.results[i].isFinal) final += t; else interim += t;
   659→      }
   660→      if (interim && state === 'listening') statusText.textContent = `شنیدم: «${interim}»`;
   661→      if (final) {
   662→        gotFinal = true;
   663→        stopListening(false);
   664→        runCommand(final.trim());
   665→      }
   666→    };
   667→    r.onerror = (e) => {
   668→      if (['network', 'not-allowed', 'service-not-allowed', 'audio-capture', 'language-not-supported'].includes(e.error)) {
   669→        srBroken = true;
   670→        refreshEngineUI();
   671→        /* فالبک خودکار: گوگل رایگان → GLM ابری (اگر کلید باشد) */
   672→        if (state === 'listening' && settings.sttEngine === 'auto') {
   673→          try { recActive = false; } catch (_) { /* noop */ }
   674→          if (googleReady()) startGoogleListen();
   675→          else if (glmReady()) startGlmListen();
   676→        }
   677→      }
   678→    };
   679→    r.onend = () => {
   680→      recActive = false;
   681→      if (gotFinal || srBroken) return;
   682→      if (state === 'listening') {
   683→        setState('idle');
   684→        statusText.innerHTML = IDLE_HINT;
   685→        orbIcon.setAttribute('href', '#i-mic');
   686→        sbMic.innerHTML = '<i class="dot ok"></i>میکروفون: آماده';
   687→      }
   688→    };
   689→    return r;
   690→  }

   693→  const G_MAX_MS = 12000;    // بیشینه ضبط
   694→  const G_SIL_MS = 1500;     // سکوت پایان فرمان
   695→  const G_ON = 0.013;        // آستانه شروع حرف (RMS)
   696→  const G_IDLE_MS = 8000;    // اگر هیچ حرفی نشنید
   697→  let gRec = null, gMaxT = null;

   699→  function downsampleF32(f32, from, to) {
   700→    if (from === to) return f32;
   701→    const ratio = from / to;
   702→    const len = Math.max(1, Math.floor(f32.length / ratio));
   703→    const out = new Float32Array(len);
   704→    for (let i = 0; i < len; i++) {
   705→      const pos = i * ratio;
   706→      const i0 = Math.floor(pos);
   707→      const frac = pos - i0;
   708→      const s0 = f32[i0] || 0;
   709→      const s1 = f32[i0 + 1] || s0;
   710→      out[i] = s0 + (s1 - s0) * frac;
   711→    }
   712→    return out;
   713→  }
   714→
   715→  function f32ToI16(f32) {
   716→    const out = new Int16Array(f32.length);
   717→    for (let i = 0; i < f32.length; i++) {
   718→      const v = Math.max(-1, Math.min(1, f32[i]));
   719→      out[i] = v < 0 ? v * 32768 : v * 32767;
   720→    }
   721→    return out;
   722→  }

   724→  function startGoogleListen() {
   725→    if (!googleReady()) { noEngine('موتور گوگل فقط داخل نرم‌افزار فعال است'); return; }
   726→    attachMic().then((ok) => {
   727→      if (!ok) { noEngine('میکروفون در دسترس نیست'); return; }
   728→      try {
   729→        const src = audioCtx.createMediaStreamSource(micStream);
   730→        const proc = audioCtx.createScriptProcessor(4096, 1, 1);
   731→        const sink = audioCtx.createGain();
   732→        sink.gain.value = 0; // بی‌صدا — فقط برای پردازش
   733→        src.connect(proc);
   734→        proc.connect(sink);
   735→        sink.connect(audioCtx.destination);
   736→        gRec = { src, proc, sink, chunks: [], spoke: false, lastVoice: 0, started: Date.now(), busy: false };
   737→        proc.onaudioprocess = (e) => {
   738→          if (!gRec || gRec.busy) return;
   739→          const f = e.inputBuffer.getChannelData(0);
   740→          gRec.chunks.push(new Float32Array(f));
   741→          let sum = 0, n = 0;
   742→          for (let i = 0; i < f.length; i += 4) { sum += f[i] * f[i]; n++; }
   743→          const rms = Math.sqrt(sum / Math.max(1, n));
   744→          const now = Date.now();
   745→          if (rms > G_ON) {
   746→            gRec.spoke = true;
   747→            gRec.lastVoice = now;
   748→            if (state === 'listening') statusText.textContent = 'شنیدم… بعد از سکوت، گوگل تبدیلش می‌کند';
   749→          } else if (gRec.spoke && now - gRec.lastVoice > G_SIL_MS) {
   750→            stopGoogleRec();
   751→          } else if (!gRec.spoke && now - gRec.started > G_IDLE_MS) {
   752→            stopGoogleRec();
   753→          }
   754→        };
   755→        statusText.textContent = 'در حال گوش دادن (گوگل)… فرمانت را بگو';
   756→        gMaxT = setTimeout(() => stopGoogleRec(), G_MAX_MS);
   757→      } catch (_) {
   758→        gRec = null;
   759→        noEngine('شروع ضبط گوگل ممکن نشد');
   760→      }
   761→    });
   762→  }

   764→  function stopGoogleRec() {

   772→    if (g.busy) return;

   774→    const totalMs = (g.chunks.length * 4096 * 1000) / (audioCtx ? audioCtx.sampleRate : 48000);
   775→    if (!g.spoke || totalMs < 350) {
   776→      statusText.textContent = 'صدایی نشنیدم؛ دوباره امتحان کن';
   777→      setTimeout(() => { if (state === 'listening' || state === 'processing') { setState('idle'); statusText.innerHTML = IDLE_HINT; orbIcon.setAttribute('href', '#i-mic'); sbMic.innerHTML = '<i class="dot ok"></i>میکروفون: آماده'; } }, 1500);
   778→      return;
   779→    }
   780→    setState('processing');
   781→    statusText.textContent = 'در حال تبدیل گفتار با گوگل…';
   782→    const merged = new Float32Array(g.chunks.reduce((a, c) => a + c.length, 0));
   783→    let off = 0;
   784→    for (const c of g.chunks) { merged.set(c, off); off += c.length; }
   785→    const rate = (audioCtx && audioCtx.sampleRate) || 48000;
   786→    const pcm16 = f32ToI16(downsampleF32(merged, rate, 16000));

   796→          toast((r && r.error) || 'گوگل پاسخی نداد', '#i-info');
   797→        }
   798→      })
   799→      .catch(() => {
   800→        setState('idle');
   801→        statusText.textContent = 'اتصال به گوگل برقرار نشد — اینترنت/فیلترشکن را چک کن';
   802→        orbIcon.setAttribute('href', '#i-mic');
   803→        sbMic.innerHTML = '<i class="dot ok"></i>میکروفون: آماده';
