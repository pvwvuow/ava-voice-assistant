   670→  function makeRec() {
   671→    const r = new SRC();
   672→    r.lang = 'fa-IR';
   673→    r.interimResults = true;
   674→    r.continuous = false;
   675→    r.onresult = (e) => {

   678→        const t = e.results[i][0].transcript;

   682→      if (final) {
   683→        gotFinal = true;
   684→        stopListening(false);
   685→        runCommand(final.trim());

   687→    };
   688→    r.onerror = (e) => {

   695→          if (googleReady()) startGoogleListen();
   696→          else if (glmReady()) startGlmListen();
   697→        }
   698→      }
   699→    };
   700→    r.onend = () => {
   701→      recActive = false;
   702→      if (gotFinal || srBroken) return;
   703→      if (state === 'listening') {
   704→        setState('idle');

   706→        orbIcon.setAttribute('href', '#i-mic');
   707→        sbMic.innerHTML = '<i class="dot ok"></i>میکروفون: آماده';

   713→  /* --- موتور رایگان گوگل: ضبط PCM + تشخیص سکوت + ارسال به سرور گوگل --- */
   714→  const G_MAX_MS = 12000;    // بیشینه ضبط
   715→  const G_SIL_MS = 1500;     // سکوت پایان فرمان
   716→  const G_ON = 0.013;        // آستانه شروع حرف (RMS)
   717→  const G_IDLE_MS = 8000;    // اگر هیچ حرفی نشنید
   718→  let gRec = null, gMaxT = null;
   719→
   720→  function downsampleF32(f32, from, to) {
   721→    if (from === to) return f32;
   722→    const ratio = from / to;
   723→    const len = Math.max(1, Math.floor(f32.length / ratio));
   724→    const out = new Float32Array(len);
   725→    for (let i = 0; i < len; i++) {
   726→      const pos = i * ratio;
   727→      const i0 = Math.floor(pos);
   728→      const frac = pos - i0;
   729→      const s0 = f32[i0] || 0;
   730→      const s1 = f32[i0 + 1] || s0;
   731→      out[i] = s0 + (s1 - s0) * frac;
   732→    }
   733→    return out;
   734→  }
   735→
   736→  function f32ToI16(f32) {
   737→    const out = new Int16Array(f32.length);
   738→    for (let i = 0; i < f32.length; i++) {
   739→      const v = Math.max(-1, Math.min(1, f32[i]));
   740→      out[i] = v < 0 ? v * 32768 : v * 32767;
   741→    }
   742→    return out;
   743→  }

   745→  function startGoogleListen() {
   746→    if (!googleReady()) { noEngine('موتور گوگل فقط داخل نرم‌افزار فعال است'); return; }
   747→    attachMic().then((ok) => {
   748→      if (!ok) { noEngine('میکروفون در دسترس نیست'); return; }
   749→      try {
   750→        const src = audioCtx.createMediaStreamSource(micStream);
   751→        const proc = audioCtx.createScriptProcessor(4096, 1, 1);
   752→        const sink = audioCtx.createGain();
   753→        sink.gain.value = 0; // بی‌صدا — فقط برای پردازش
   754→        src.connect(proc);
   755→        proc.connect(sink);
   756→        sink.connect(audioCtx.destination);
   757→        gRec = { src, proc, sink, chunks: [], spoke: false, lastVoice: 0, started: Date.now(), busy: false };
   758→        proc.onaudioprocess = (e) => {
   759→          if (!gRec || gRec.busy) return;
   760→          const f = e.inputBuffer.getChannelData(0);
   761→          gRec.chunks.push(new Float32Array(f));
   762→          let sum = 0, n = 0;
   763→          for (let i = 0; i < f.length; i += 4) { sum += f[i] * f[i]; n++; }
   764→          const rms = Math.sqrt(sum / Math.max(1, n));
   765→          const now = Date.now();
   766→          if (rms > G_ON) {
   767→            gRec.spoke = true;
   768→            gRec.lastVoice = now;
   769→            if (state === 'listening') statusText.textContent = 'شنیدم… بعد از سکوت، گوگل تبدیلش می‌کند';
   770→          } else if (gRec.spoke && now - gRec.lastVoice > G_SIL_MS) {
   771→            stopGoogleRec();
   772→          } else if (!gRec.spoke && now - gRec.started > G_IDLE_MS) {
   773→            stopGoogleRec();
   774→          }
   775→        };
   776→        statusText.textContent = 'در حال گوش دادن (گوگل)… فرمانت را بگو';
   777→        gMaxT = setTimeout(() => stopGoogleRec(), G_MAX_MS);
   778→      } catch (_) {
   779→        gRec = null;
   780→        noEngine('شروع ضبط گوگل ممکن نشد');
   781→      }
   782→    });
   783→  }

   785→  function stopGoogleRec() {
   786→    clearTimeout(gMaxT); gMaxT = null;
   787→    if (!gRec) return;
   788→    const g = gRec;
   789→    gRec = null;
   790→    try { g.proc.disconnect(); } catch (_) { /* noop */ }
   791→    try { g.src.disconnect(); } catch (_) { /* noop */ }
   792→    try { g.sink.disconnect(); } catch (_) { /* noop */ }
   793→    if (g.busy) return;
   794→    g.busy = true;
   795→    const totalMs = (g.chunks.length * 4096 * 1000) / (audioCtx ? audioCtx.sampleRate : 48000);
   796→    if (!g.spoke || totalMs < 350) {
   797→      statusText.textContent = 'صدایی نشنیدم؛ دوباره امتحان کن';
   798→      setTimeout(() => { if (state === 'listening' || state === 'processing') { setState('idle'); statusText.innerHTML = IDLE_HINT; orbIcon.setAttribute('href', '#i-mic'); sbMic.innerHTML = '<i class="dot ok"></i>میکروفون: آماده'; } }, 1500);
   799→      return;
   800→    }
   801→    setState('processing');
   802→    statusText.textContent = 'در حال تبدیل گفتار با گوگل…';
   803→    const merged = new Float32Array(g.chunks.reduce((a, c) => a + c.length, 0));
   804→    let off = 0;
   805→    for (const c of g.chunks) { merged.set(c, off); off += c.length; }
   806→    const rate = (audioCtx && audioCtx.sampleRate) || 48000;
   807→    const pcm16 = f32ToI16(downsampleF32(merged, rate, 16000));
   808→    bridge.stt.google({ pcm: new Uint8Array(pcm16.buffer), rate: 16000, key: settings.googleKey || '', lang: 'fa-IR' })
   809→      .then((r) => {
   810→        if (r && r.ok && r.text) {
   811→          runCommand(r.text.trim());
   812→        } else {
   813→          setState('idle');
   814→          statusText.textContent = 'تبدیل گوگل ممکن نشد: ' + ((r && r.error) || 'خطای نامشخص');
   815→          orbIcon.setAttribute('href', '#i-mic');
   816→          sbMic.innerHTML = '<i class="dot ok"></i>میکروفون: آماده';
   817→          toast((r && r.error) || 'گوگل پاسخی نداد', '#i-info');
   818→        }
   819→      })
   820→      .catch(() => {
   821→        setState('idle');
   822→        statusText.textContent = 'اتصال به گوگل برقرار نشد — اینترنت/فیلترشکن را چک کن';
   823→        orbIcon.setAttribute('href', '#i-mic');
   824→        sbMic.innerHTML = '<i class="dot ok"></i>میکروفون: آماده';
   825→      });
   826→  }
