   236→  function rr(c, x, y, w, hgt, r) {
   237→    if (c.roundRect) { c.beginPath(); c.roundRect(x, y, w, hgt, r); return; }
   238→    c.beginPath();
   239→    c.moveTo(x + r, y);
   240→    c.arcTo(x + w, y, x + w, y + hgt, r);
   241→    c.arcTo(x + w, y + hgt, x, y + hgt, r);
   242→    c.arcTo(x, y + hgt, x, y, r);
   243→    c.arcTo(x, y, x + w, y, r);
   244→    c.closePath();
   245→  }
   246→  function frame() {
   247→    t0 += 0.016;
   248→    const target = state === 'listening' ? 0.88 : state === 'processing' ? 0.42 : state === 'success' ? 0.55 : 0.15;
   249→    energy += (target - energy) * 0.05;
   250→    ctx.clearRect(0, 0, W, H);
   251→    const mid = H / 2;
   252→    const gap = 4;
   253→    const bw = Math.max(2, Math.min(4.5, (W - (N - 1) * gap) / N));
   254→    const startX = (W - (N * bw + (N - 1) * gap)) / 2;
   255→    for (let i = 0; i < N; i++) {
   256→      const env = Math.sin((Math.PI * i) / (N - 1));
   257→      let lvl;
   258→      if (micLive && micData) {
   259→        /* صدای واقعی میکروفون: تبدیل طیف فرکانسی به ۵۲ میله */
   260→        const bins = Math.floor(micData.length * 0.72);
   261→        const bi = Math.min(micData.length - 1, Math.floor(Math.pow(i / N, 1.55) * bins));
   262→        const raw = micData[bi] / 255;
   263→        lvl = Math.max(0.05, Math.min(1, raw * 1.6 * (0.35 + 0.65 * env)));
   264→      } else {
   265→        const n =
   266→          Math.sin(t0 * 2.1 + i * 0.55) * 0.5 +
   267→          Math.sin(t0 * 3.7 + i * 1.3) * 0.3 +
   268→          Math.sin(t0 * 0.7 + i * 0.21) * 0.2;
   269→        const amp = energy * env * (0.32 + 0.68 * Math.abs(n));
   270→        const jitter = energy > 0.2 ? Math.random() * 0.13 * energy : 0;
   271→        lvl = Math.max(0.04, Math.min(1, amp + jitter));
   272→      }
   273→      levels[i] += (lvl - levels[i]) * 0.25;
   274→      const bh = Math.max(3, levels[i] * (H - 8));
   275→      const g = ctx.createLinearGradient(0, mid - bh / 2, 0, mid + bh / 2);
   276→      g.addColorStop(0, 'rgba(52, 211, 153, 0.95)');
   277→      g.addColorStop(0.5, 'rgba(16, 185, 129, 0.85)');
   278→      g.addColorStop(1, 'rgba(13, 148, 136, 0.9)');
   279→      ctx.fillStyle = g;
   280→      rr(ctx, startX + i * (bw + gap), mid - bh / 2, bw, bh, bw / 2);
   281→      ctx.fill();
   282→    }
   283→    requestAnimationFrame(frame);
   284→  }

   395→    try {

   397→      if (!AC) return;

   402→      o.type = 'sine'; o.frequency.value = 880;
   403→      g.gain.setValueAtTime(0.001, ac.currentTime);
   404→      g.gain.exponentialRampToValueAtTime(0.2, ac.currentTime + 0.05);
   405→      g.gain.exponentialRampToValueAtTime(0.001, ac.currentTime + 0.9);

   511→  function makeRec() {
   512→    const r = new SRC();
   513→    r.lang = 'fa-IR';
   514→    r.interimResults = true;
   515→    r.continuous = false;
   516→    r.onresult = (e) => {
   517→      let interim = '', final = '';
   518→      for (let i = e.resultIndex; i < e.results.length; i++) {
   519→        const t = e.results[i][0].transcript;
   520→        if (e.results[i].isFinal) final += t; else interim += t;
   521→      }
   522→      if (interim && state === 'listening') statusText.textContent = `شنیدم: «${interim}»`;
   523→      if (final) {
   524→        gotFinal = true;
   525→        stopListening(false);
   526→        runCommand(final.trim());
   527→      }
   528→    };
   529→    r.onerror = (e) => {
   530→      if (['network', 'not-allowed', 'service-not-allowed', 'audio-capture'].includes(e.error)) {
   531→        srBroken = true;
   532→        updateEngine();
   533→      }
   534→    };
   535→    r.onend = () => {
   536→      recActive = false;
   537→      if (gotFinal || srBroken) return;
   538→      if (state === 'listening') {
   539→        setState('idle');
   540→        statusText.innerHTML = IDLE_HINT;
   541→        orbIcon.setAttribute('href', '#i-mic');
   542→        sbMic.innerHTML = '<i class="dot ok"></i>میکروفون: آماده';
   543→      }
   544→    };
   545→    return r;
   546→  }
