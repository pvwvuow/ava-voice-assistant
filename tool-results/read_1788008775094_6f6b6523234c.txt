   330→  const N = 52;
   331→  const levels = new Array(N).fill(0.1);
   332→  let t0 = 0, energy = 0.1;
   333→  function rr(c, x, y, w, hgt, r) {
   334→    if (c.roundRect) { c.beginPath(); c.roundRect(x, y, w, hgt, r); return; }
   335→    c.beginPath();
   336→    c.moveTo(x + r, y);
   337→    c.arcTo(x + w, y, x + w, y + hgt, r);
   338→    c.arcTo(x + w, y + hgt, x, y + hgt, r);
   339→    c.arcTo(x, y + hgt, x, y, r);
   340→    c.arcTo(x, y, x + w, y, r);
   341→    c.closePath();
   342→  }
   343→  function frame() {
   344→    t0 += 0.016;
   345→    const target = state === 'listening' ? 0.88 : state === 'processing' ? 0.42 : state === 'success' ? 0.55 : 0.15;
   346→    energy += (target - energy) * 0.05;
   347→    ctx.clearRect(0, 0, W, H);
   348→    const mid = H / 2;
   349→    const gap = 4;
   350→    const bw = Math.max(2, Math.min(4.5, (W - (N - 1) * gap) / N));
   351→    const startX = (W - (N * bw + (N - 1) * gap)) / 2;
   352→    for (let i = 0; i < N; i++) {
   353→      const env = Math.sin((Math.PI * i) / (N - 1));
   354→      let lvl;
   355→      if (micData) {
   356→        /* صدای واقعی میکروفون: تبدیل طیف فرکانسی به ۵۲ میله — همیشه واقعی */
   357→        const bins = Math.floor(micData.length * 0.72);
   358→        const bi = Math.min(micData.length - 1, Math.floor(Math.pow(i / N, 1.55) * bins));
   359→        const raw = micData[bi] / 255;
   360→        lvl = Math.max(0.05, Math.min(1, raw * 1.6 * (0.35 + 0.65 * env)));
   361→      } else {
   362→        const n =
   363→          Math.sin(t0 * 2.1 + i * 0.55) * 0.5 +
   364→          Math.sin(t0 * 3.7 + i * 1.3) * 0.3 +
   365→          Math.sin(t0 * 0.7 + i * 0.21) * 0.2;
   366→        const amp = energy * env * (0.32 + 0.68 * Math.abs(n));
   367→        const jitter = energy > 0.2 ? Math.random() * 0.13 * energy : 0;
   368→        lvl = Math.max(0.04, Math.min(1, amp + jitter));
   369→      }
   370→      levels[i] += (lvl - levels[i]) * 0.25;
   371→      const bh = Math.max(3, levels[i] * (H - 8));
   372→      const g = ctx.createLinearGradient(0, mid - bh / 2, 0, mid + bh / 2);
   373→      g.addColorStop(0, 'rgba(52, 211, 153, 0.95)');
   374→      g.addColorStop(0.5, 'rgba(16, 185, 129, 0.85)');
   375→      g.addColorStop(1, 'rgba(13, 148, 136, 0.9)');
   376→      ctx.fillStyle = g;
   377→      rr(ctx, startX + i * (bw + gap), mid - bh / 2, bw, bh, bw / 2);
   378→      ctx.fill();
   379→    }
   380→    requestAnimationFrame(frame);
   381→  }

  1141→  async function runCustom(cc) {
  1142→    const act = cc.action || {};
  1143→    if (act.type === 'open_url') {
  1144→      if (bridge && bridge.system && bridge.system.openUrl) {
  1145→        const r = await bridge.system.openUrl(act.value);
  1146→        return r && r.ok ? `«${cc.title}» باز شد.` : 'باز کردن لینک ممکن نشد.';
  1147→      }
  1148→      window.open(act.value, '_blank');
  1149→      return `«${cc.title}» باز شد (در مرورگر).`;
  1150→    }
  1151→    if (act.type === 'run') {
  1152→      if (!canRun) return 'اجرای واقعی فقط داخل نرم‌افزار ویندوزی انجام می‌شود.';
  1153→      const r = await bridge.system.run(act.value);
  1154→      return r && r.ok ? `«${cc.title}» انجام شد.` : `اجرا نشد: ${(r && r.error) || 'خطای نامشخص'}`;
  1155→    }
  1156→    if (act.type === 'ps') {
  1157→      if (!bridge || !bridge.custom) return 'اجرای اسکریپت فقط داخل نرم‌افزار ویندوزی انجام می‌شود.';
  1158→      const okGo = await askConfirm({
  1159→        title: 'اجرای فرمان سفارشی',
  1160→        text: `اسکریپت PowerShell زیر برای فرمان «${cc.title}» ذخیره شده. اجرا شود؟`,
  1161→        code: act.value,
  1162→      });
  1163→      if (!okGo) return 'بی‌خیال؛ اجرا نشد.';
  1164→      const r = await bridge.custom.run(act.value);
  1165→      if (r && r.ok) return (r.out ? `انجام شد: ${r.out}` : 'انجام شد.') + '';
  1166→      return `اجرا نشد: ${(r && r.error) || 'خطای نامشخص'}`;
  1167→    }
  1168→    return 'نوع فرمان سفارشی پشتیبانی نمی‌شود.';
  1169→  }
