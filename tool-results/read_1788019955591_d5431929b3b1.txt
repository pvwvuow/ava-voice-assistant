    34→    34→function serveAvaFile(reqUrl) {
    35→    35→  try {
    36→    36→    const u = new URL(reqUrl);
    37→    37→    if (u.host !== 'app') return new Response('not found', { status: 404 });
    38→    38→    const root = __dirname;
    39→    39→    const rel = decodeURIComponent(u.pathname).replace(/^\/+/, '');
    40→    40→    const file = path.normalize(path.join(root, rel || 'renderer/index.html'));
    41→    41→    if (!file.startsWith(root)) return new Response('forbidden', { status: 403 });
    42→    42→    const data = fs.readFileSync(file);
    43→    43→    const type = MIME[path.extname(file).toLowerCase()] || 'application/octet-stream';
    44→    44→    const headers = {
    45→    45→      'Content-Type': type,
    46→    46→      'Access-Control-Allow-Origin': '*',
    47→    47→      'Cache-Control': 'no-cache',
    48→    48→    };
    49→    49→    /* فقط برای سند اصلی: COOP/COEP → crossOriginIsolated=true →
    50→    50→       SharedArrayBuffer در دسترس → WASM چندنخی و تشخیص گفتار ۲ تا ۴ برابر سریع‌تر.
    51→    51→       بقیه فایل‌ها (و پاپ‌آپ‌های https مثل ورود گوگل) تحت تأثیر قرار نمی‌گیرند. */
    52→    52→    if (file.endsWith('index.html')) {
    53→    53→      headers['Cross-Origin-Opener-Policy'] = 'same-origin';
    54→    54→      headers['Cross-Origin-Embedder-Policy'] = 'require-corp';
    55→    55→    }
    56→    56→    return new Response(data, { status: 200, headers });
    57→    57→  } catch (_) {
    58→    58→    return new Response('not found', { status: 404, headers: { 'Access-Control-Allow-Origin': '*' } });
    59→    59→  }
    60→    60→}

    78→    78→function cpuSample() {
    79→    79→  const cpus = os.cpus();
    80→    80→  let idle = 0;
    81→    81→  let total = 0;
    82→    82→  for (const c of cpus) {
    83→    83→    for (const k of Object.keys(c.times)) total += c.times[k];
    84→    84→    idle += c.times.idle;
    85→    85→  }
    86→    86→  return { idle, total };
    87→    87→}
    88→    88→let prevCpu = cpuSample();
    89→    89→
    90→    90→function cpuUsage() {
    91→    91→  const cur = cpuSample();
    92→    92→  const idleD = cur.idle - prevCpu.idle;
    93→    93→  const totalD = cur.total - prevCpu.total;
    94→    94→  prevCpu = cur;
    95→    95→  if (totalD <= 0) return 0;
    96→    96→  return Math.max(0, Math.min(100, Math.round((1 - idleD / totalD) * 100)));
    97→    97→}
