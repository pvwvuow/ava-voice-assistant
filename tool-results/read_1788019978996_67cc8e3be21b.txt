   834→  function downsampleF32(f32, from, to) {
   835→    if (from === to) return f32;
   836→    const ratio = from / to;
   837→    const len = Math.max(1, Math.floor(f32.length / ratio));
   838→    const out = new Float32Array(len);
   839→    for (let i = 0; i < len; i++) {
   840→      const pos = i * ratio;
   841→      const i0 = Math.floor(pos);
   842→      const frac = pos - i0;
   843→      const s0 = f32[i0] || 0;
   844→      const s1 = f32[i0 + 1] || s0;
   845→      out[i] = s0 + (s1 - s0) * frac;
   846→    }
   847→    return out;
   848→  }

   850→  function f32ToI16(f32) {
   851→    const out = new Int16Array(f32.length);
   852→    for (let i = 0; i < f32.length; i++) {
   853→      const v = Math.max(-1, Math.min(1, f32[i]));
   854→      out[i] = v < 0 ? v * 32768 : v * 32767;
   855→    }
   856→    return out;
   857→  }

   859→  function startGoogleListen() {
   860→    if (!googleReady()) { noEngine('موتور گوگل فقط داخل نرم‌افزار فعال است'); return; }
   861→    attachMic().then((ok) => {
   862→      if (!ok) { noEngine('میکروفون در دسترس نیست'); return; }
   863→      try {
   864→        const src = audioCtx.createMediaStreamSource(micStream);
   865→        const proc = audioCtx.createScriptProcessor(4096, 1, 1);
   866→        const sink = audioCtx.createGain();
   867→        sink.gain.value = 0; // بی‌صدا — فقط برای پردازش
   868→        src.connect(proc);
   869→        proc.connect(sink);
   870→        sink.connect(audioCtx.destination);
   871→        gRec = { src, proc, sink, chunks: [], spoke: false, lastVoice: 0, started: Date.now(), busy: false };
   872→        proc.onaudioprocess = (e) => {
   873→          if (!gRec || gRec.busy) return;
   874→          const f = e.inputBuffer.getChannelData(0);
   875→          gRec.chunks.push(new Float32Array(f));
   876→          let sum = 0, n = 0;
   877→          for (let i = 0; i < f.length; i += 4) { sum += f[i] * f[i]; n++; }
   878→          const rms = Math.sqrt(sum / Math.max(1, n));
   879→          const now = Date.now();
   880→          if (rms > G_ON) {
   881→            gRec.spoke = true;
   882→            gRec.lastVoice = now;
   883→            if (state === 'listening') statusText.textContent = 'شنیدم… بعد از سکوت، گوگل تبدیلش می‌کند';
   884→          } else if (gRec.spoke && now - gRec.lastVoice > G_SIL_MS) {
   885→            stopGoogleRec();
   886→          } else if (!gRec.spoke && now - gRec.started > G_IDLE_MS) {
   887→            stopGoogleRec();
   888→          }
   889→        };
   890→        statusText.textContent = 'در حال گوش دادن (گوگل)… فرمانت را بگو';
   891→        gMaxT = setTimeout(() => stopGoogleRec(), G_MAX_MS);
   892→      } catch (_) {
   893→        gRec = null;
   894→        noEngine('شروع ضبط گوگل ممکن نشد');
   895→      }
   896→    });
   897→  }
